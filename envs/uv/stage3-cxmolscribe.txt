# uv-managed interpreter. From the repository root:
#
#   uv venv .venv-ms --python 3.11
#   uv pip install --python .venv-ms/bin/python -r envs/uv/stage3-cxmolscribe.txt
#   uv pip install --python .venv-ms/bin/python --no-deps -e cxmolscribe-wd/MolScribe
#
# MolScribe itself is NOT listed here (same as the yml): it is installed with
# --no-deps from cxmolscribe-wd/MolScribe, because letting the resolver see
# MolScribe's own install_requires pulls an incompatible torch. See
# envs/README.md, section "MolScribe / CXMolScribe".
#
# Reconstructed from the working `2ms` environment on the CRC cluster; the raw
# capture is kept at envs/legacy/cxmolscribe_conda_explicit.txt.

# --- PyTorch ------------------------------------------------------------
# Linux x86_64 PyPI wheels for 2.7.1 are CUDA 12.6 builds, matching the
# reference environment. The Linux aarch64 wheels are CPU-only builds.
torch==2.7.1
torchvision==0.22.1

# --- MolScribe decoder --------------------------------------------------
# MolScribe's decoder is built on OpenNMT-py 2.2.0, which in turn hard-pins
# the legacy torchtext 0.5.0. Both pins are load-bearing; newer versions of
# either break the checkpoint. torchtext 0.5.0 has no wheel and builds from
# an sdist, so this is the slowest part of the install.
#   [uv note] The sentence above is carried over from the yml but is not
#   accurate: PyPI does publish torchtext-0.5.0-py3-none-any.whl (pure
#   Python, 73 kB), and uv installs it in seconds on any architecture.
#   Nothing is compiled. It is still imported at inference time: molscribe's
#   transformer decoder imports onmt, and onmt 2.2.0 imports torchtext.
OpenNMT-py==2.2.0
torchtext==0.5.0
pyonmttok==1.37.1
sentencepiece==0.2.0
configargparse==1.7.1

# --- Indigo on Linux aarch64 (not in the yml) ---------------------------
# molscribe/dataset.py -- imported by molscribe/interface.py, hence by any
# `from molscribe import MolScribe` -- uses the Indigo toolkit that MolScribe
# vendors under molscribe/indigo/. The vendored native libraries are x86_64
# only (molscribe/indigo/lib/Linux/x64/*.so). epam.indigo ships the same
# toolkit, with an API-compatible Python binding, as manylinux2014 aarch64
# wheels. patches/0001-molscribe-indigo-aarch64-fallback.patch makes
# molscribe use it when the vendored library is not loadable on this machine.
# Harmless on x86_64, where the vendored copy is still used.
epam.indigo==1.46.0
